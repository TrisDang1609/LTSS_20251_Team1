cmake_minimum_required(VERSION 3.22)

IF(NOT CMAKE_BUILD_TYPE)
  SET(CMAKE_BUILD_TYPE Release)
ENDIF()

MESSAGE("Build type: " ${CMAKE_BUILD_TYPE})

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS}  -Wall  -O3 -march=native ")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall   -O3 -march=native")

# Check C++17 support (upgraded from C++14 for CUDA 13.0 compatibility)
include(CheckCXXCompilerFlag)
CHECK_CXX_COMPILER_FLAG("-std=c++17" COMPILER_SUPPORTS_CXX17)
if(COMPILER_SUPPORTS_CXX17)
   set(CMAKE_CXX_STANDARD 17)
   set(CMAKE_CXX_STANDARD_REQUIRED ON)
   add_definitions(-DCOMPILEDWITHC17)
   message(STATUS "Using C++17 standard.")
else()
   message(FATAL_ERROR "The compiler ${CMAKE_CXX_COMPILER} has no C++17 support. Please use a different C++ compiler.")
endif()

LIST(APPEND CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/cmake_modules)

# CUDA Support - Include CUDA extension module
# ============================================================================
option(USE_CUDA "Enable CUDA GPU acceleration" ON)

    message(STATUS "")
    message(STATUS "========================================")
    message(STATUS "  Enabling CUDA GPU Acceleration")
    message(STATUS "========================================")
    
    # Include the CUDA configuration
    include(${CMAKE_SOURCE_DIR}/cmake/CMakeLists_cuda.cmake)
    
    # Link CUDA library to main ORB-SLAM3 library
    link_cuda_to_orbslam(${PROJECT_NAME})
    
    message(STATUS "CUDA support enabled successfully")
    message(STATUS "========================================")
    message(STATUS "")
endif()
